def find_Bohm_force_at_position_const_sigma(N,pos,x,y,z,packet_width,e_mass,hbar):
    
    force_x = 0
    force_y = 0
    force_z = 0
    
    targ_pos = np.zeros((1,3))
    targ_pos[0,0] = x
    targ_pos[0,1] = y
    targ_pos[0,2] = z
    
    sum_packets = 0
    
    first_order_x_term = 0
    first_order_y_term = 0
    first_order_z_term = 0
    
    second_order_x_term = 0
    second_order_y_term = 0
    second_order_z_term = 0
    
    theta_x = 0
    theta_y = 0
    theta_z = 0

    phi = 0
    
    rho_x = 0 
    
    rho_x1 = 0
    rho_x2 = 0
    rho_x3 = 0
    
    rho_y = 0
    
    rho_y1 = 0
    rho_y3 = 0
    
    rho_z = 0
    
    rho_z1 = 0

    for j in range(int(N)):
        #print('delx = '+str(targ_pos[0,0] - pos[j,0]))
        #print('dely = '+str(targ_pos[0,1] - pos[j,1]))
        #print('delz = '+str(targ_pos[0,2] - pos[j,2]))
        sum_packets += np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2)))
        first_order_x_term += ((targ_pos[0,0]-pos[j,0])/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))*(1 - ((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(packet_width**2))
        first_order_y_term += ((targ_pos[0,1]-pos[j,1])/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))*(1 - ((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(packet_width**2))
        first_order_z_term += ((targ_pos[0,2]-pos[j,2])/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))*(1 - ((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(packet_width**2))
        theta_x += ((targ_pos[0,0]-pos[j,0])/(packet_width**2))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        theta_y += ((targ_pos[0,1]-pos[j,1])/(packet_width**2))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        theta_z += ((targ_pos[0,2]-pos[j,2])/(packet_width**2))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        phi += (((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(packet_width**4))*np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2)))
        rho_x1 += (((targ_pos[0,0] - pos[j,0])**2)/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        rho_x2 += (((targ_pos[0,0] - pos[j,0])*(targ_pos[0,1] - pos[j,1]))/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        rho_x3 += (((targ_pos[0,0] - pos[j,0])*(targ_pos[0,2] - pos[j,2]))/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        rho_y1 += (((targ_pos[0,1] - pos[j,1])**2)/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        rho_y3 += (((targ_pos[0,1] - pos[j,1])*(targ_pos[0,2] - pos[j,2]))/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))
        rho_z1 += (((targ_pos[0,2] - pos[j,2])**2)/(packet_width**4))*(np.exp(-((targ_pos[0,0] - pos[j,0])**2 + (targ_pos[0,1] - pos[j,1])**2 + (targ_pos[0,2] - pos[j,2])**2)/(2*(packet_width**2))))

    #print('sum_packets = '+str(sum_packets))
    
    first_order_x_term *= (1/sum_packets)
    first_order_y_term *= (1/sum_packets)
    first_order_z_term *= (1/sum_packets)
    #print('first_order_x_term = '+str(first_order_x_term))
    #print('first_order_y_term = '+str(first_order_y_term))
    #print('first_order_z_term = '+str(first_order_z_term))
        
    rho_x = theta_x*rho_x1 + theta_y*rho_x2 + theta_z*rho_x3
    #print('rho_x = '+str(rho_x))
    
    rho_y = theta_x*rho_x2 + theta_y*rho_y1 + theta_z*rho_y3
    #print('rho_y = '+str(rho_y))
    
    rho_z = theta_x*rho_x3 + theta_y*rho_y3 + theta_z*rho_z1
    #print('rho_z = '+str(rho_z))
    
    #print('phi = '+str(phi))
    
    second_order_x_term = phi*theta_x + rho_x
    second_order_y_term = phi*theta_y + rho_y
    second_order_z_term = phi*theta_z + rho_z
    
    second_order_x_term *= (1/(sum_packets**2))
    second_order_y_term *= (1/(sum_packets**2))
    second_order_z_term *= (1/(sum_packets**2))
    #print('second_order_x_term = '+str(second_order_x_term))
    #print('second_order_y_term = '+str(second_order_y_term))
    #print('second_order_z_term = '+str(second_order_z_term))
      
    gamma = theta_x**2 + theta_y**2 + theta_z**2
    #print('gamma = '+str(gamma))
    
    third_order_x_term = -theta_x*gamma
    third_order_y_term = -theta_y*gamma
    third_order_z_term = -theta_z*gamma
    
    third_order_x_term *= (1/(sum_packets**3))
    third_order_y_term *= (1/(sum_packets**3))
    third_order_z_term *= (1/(sum_packets**3))
    
    #print('third_order_x_term = '+str(third_order_x_term))
    #print('third_order_y_term = '+str(third_order_y_term))
    #print('third_order_z_term = '+str(third_order_z_term)+'\n')
    
    force_x = ((hbar**2)/(4*e_mass))*(first_order_x_term + second_order_x_term + third_order_x_term)
    force_y = ((hbar**2)/(4*e_mass))*(first_order_y_term + second_order_y_term + third_order_y_term)
    force_z = ((hbar**2)/(4*e_mass))*(first_order_z_term + second_order_z_term + third_order_z_term)
        
    force = np.array([force_x,force_y,force_z])
#    #print('force at position '+str(targ_pos)+' = '+str(force))
    return force
