@njit
def find_local_Bohm_force(N,pos,x,y,z,e_mass,hbar,eta,sigma_free,sigma_core,core_electrons,gamma_factor):

    # Function to compute the 'local' Bohm force, i.e. the local derivative of the Bohm potential at a point in space in the box. This does not
    # include terms calculated for the full Hamiltonian Bohm force expression.
    
    force_x = 0
    force_y = 0
    force_z = 0
    
    targ_pos = np.zeros((1,3))
    targ_pos[0,0] = x
    targ_pos[0,1] = y
    targ_pos[0,2] = z
    
    sum_packets = 0
    
    first_order_x_term = 0
    first_order_y_term = 0
    first_order_z_term = 0
    
    second_order_x_term = 0
    second_order_y_term = 0
    second_order_z_term = 0
    
    theta_x = 0
    theta_y = 0
    theta_z = 0

    psi = 0
    
    rho_x = 0 
    
    rho_x1 = 0
    rho_x2 = 0
    rho_x3 = 0
    
    rho_y = 0
    
    rho_y1 = 0
    rho_y3 = 0
    
    rho_z = 0
    
    rho_z1 = 0
    
    ion_zeta = core_electrons/(sigma_core**3)
    free_zeta = 1/(sigma_free**3)
    
    sigma_f2 = sigma_free**2
    sigma_c2 = sigma_core**2
    count = 0.0
    
    # standard Bohm loop

    for j in range(int(N)):
        delx = targ_pos[0,0] - pos[j,0]
        dely = targ_pos[0,1] - pos[j,1]
        delz = targ_pos[0,2] - pos[j,2]
        rsq = delx**2 + dely**2 + delz**2
        
        if pos[j,3] == 1:
            
            omega = np.exp(-(rsq/(2*sigma_c2)))
            
            sum_packets += ion_zeta*omega
            
            phi = (ion_zeta*omega)/sigma_c2
            
            first_order_x_term += ((delx)/sigma_c2)*(phi)*(5 - (rsq)/sigma_c2)
            first_order_y_term += ((dely)/sigma_c2)*(phi)*(5 - (rsq)/sigma_c2)
            first_order_z_term += ((delz)/sigma_c2)*(phi)*(5 - (rsq)/sigma_c2)
            theta_x += phi*delx
            theta_y += phi*dely
            theta_z += phi*delz
            psi += phi*((rsq/sigma_c2) - 3)
            rho_x1 += phi*(-1 + ((delx**2)/sigma_c2))
            rho_x2 += phi*((delx)*(dely))/sigma_c2
            rho_x3 += phi*((delx)*(delz))/sigma_c2
            rho_y1 += phi*(-1 + ((dely**2)/sigma_c2))
            rho_y3 += phi*((dely)*(delz))/sigma_c2
            rho_z1 += phi*(-1 + ((delz**2)/sigma_c2))
            
        else:
            
            omega = np.exp(-(rsq/(2*sigma_f2)))
            
            sum_packets += free_zeta*omega
            
            phi = (free_zeta*omega)/sigma_f2
            
            first_order_x_term += ((delx)/sigma_f2)*(phi)*(5 - (rsq)/sigma_f2)
            first_order_y_term += ((dely)/sigma_f2)*(phi)*(5 - (rsq)/sigma_f2)
            first_order_z_term += ((delz)/sigma_f2)*(phi)*(5 - (rsq)/sigma_f2)
            theta_x += phi*delx
            theta_y += phi*dely
            theta_z += phi*delz
            psi += phi*((rsq/sigma_f2) - 3)
            rho_x1 += phi*(-1 + ((delx**2)/sigma_f2))
            rho_x2 += phi*((delx)*(dely))/sigma_f2
            rho_x3 += phi*((delx)*(delz))/sigma_f2
            rho_y1 += phi*(-1 + ((dely**2)/sigma_f2))
            rho_y3 += phi*((dely)*(delz))/sigma_f2
            rho_z1 += phi*(-1 + ((delz**2)/sigma_f2))

    # for small sigma tests, include a minimum value for sigma:
        
    if sum_packets < eta:
        print('Applying eta for small sigma.')
        sum_packets = eta

    first_order_x_term *= (1/sum_packets)
    first_order_y_term *= (1/sum_packets)
    first_order_z_term *= (1/sum_packets)
        
    rho_x = theta_x*rho_x1 + theta_y*rho_x2 + theta_z*rho_x3
    
    rho_y = theta_x*rho_x2 + theta_y*rho_y1 + theta_z*rho_y3
    
    rho_z = theta_x*rho_x3 + theta_y*rho_y3 + theta_z*rho_z1
    
    second_order_x_term = psi*theta_x + rho_x
    second_order_y_term = psi*theta_y + rho_y
    second_order_z_term = psi*theta_z + rho_z
    
    second_order_x_term *= (1/(sum_packets**2))
    second_order_y_term *= (1/(sum_packets**2))
    second_order_z_term *= (1/(sum_packets**2))
      
    gamma = theta_x**2 + theta_y**2 + theta_z**2
    
    third_order_x_term = -theta_x*gamma
    third_order_y_term = -theta_y*gamma
    third_order_z_term = -theta_z*gamma
    
    third_order_x_term *= (1/(sum_packets**3))
    third_order_y_term *= (1/(sum_packets**3))
    third_order_z_term *= (1/(sum_packets**3))
    
    force_x = ((hbar**2)/(4*e_mass))*(first_order_x_term + second_order_x_term + third_order_x_term)
    force_y = ((hbar**2)/(4*e_mass))*(first_order_y_term + second_order_y_term + third_order_y_term)
    force_z = ((hbar**2)/(4*e_mass))*(first_order_z_term + second_order_z_term + third_order_z_term)
        
    force = gamma_factor*np.array([force_x,force_y,force_z])
    
    bohm_pot = -gamma_factor*(((hbar**2)/(4*e_mass))/2)*((2*psi)/sum_packets - (gamma)/sum_packets**2)

    return bohm_pot, force
